```python
import sys

MOD = 1_000_000_007

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n, m = map(int, data[:2])
    names = data[2:2 + n]

    # For each column keep the set of distinct letters seen so far
    col_sets = [set() for _ in range(m)]

    for name in names:
        # name is guaranteed to have length m
        for idx, ch in enumerate(name):
            col_sets[idx].add(ch)

    # The answer is the product of the sizes of all column sets
    result = 1
    for s in col_sets:
        result = (result * len(s)) % MOD

    print(result)


if __name__ == "__main__":
    solve()
```

using System.IO;
using System.Numerics;

class Program
{
    const long MOD = 1000000007L;
    static void Main()
    {

        int n = input.NextInt();
        int m = input.NextInt();
        int[] mask = new int[m];
        for (int i = 0; i < n; i++)
        {
            string s = input.NextToken();
            for (int j = 0; j < m; j++)
            {
                int bit = s[j] - 'A';
                mask[j] |= 1 << bit;
            }

        long result = 1;
        foreach (int bits in mask)
        {
            int cnt = PopCount(bits);
            result = (result * cnt) % MOD;
        }
        Console.WriteLine(result);
    }

    static int PopCount(int x)
    {
        int cnt = 0;
        while (x != 0)
        {
            x &= x - 1;
            cnt++;
        }
        return cnt;
    }

    class FastScanner
    {
        private readonly byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        private readonly Stream stream;

        public FastScanner(Stream stream = null)
        {
            this.stream = stream ?? Console.OpenStandardInput();
        }

        private int ReadByte()
        {
            if (ptr >= len)
            {
                len = stream.Read(buffer, 0, buffer.Length);
                ptr = 0;
                if (len == 0) return -1;
            }
            return buffer[ptr++];
        }

**Optimized C# solution**

```csharp
using System;
using System.IO;

class Program
{
    const long MOD = 1_000_000_007L;

    static void Main()
    {
        var fs = new FastScanner();
        int n = fs.NextInt();
        int m = fs.NextInt();

        // for every position keep a bit mask of letters that appear there
        long[] mask = new long[m];

        for (int i = 0; i < n; i++)
        {
            string s = fs.NextToken();          // name
            for (int j = 0; j < m; j++)
            {
                int bit = s[j] - 'A';           // 0 … 25 (letters are uppercase)
                mask[j] |= 1L << bit;
            }
        }

        long answer = 1;
        foreach (long bits in mask)
        {
            int cnt = PopCount(bits);          // number of different letters
            answer = (answer * cnt) % MOD;
        }

        Console.WriteLine(answer);
    }

    // pop-count for a 64-bit value – very fast
    static int PopCount(long x)
    {
        int cnt = 0;
        while (x != 0)
        {
            x &= x - 1;
            cnt++;
        }
        return cnt;
    }

    /* ---------- fast scanner ----------
       reads integers and tokens from stdin
       -----------------------------------*/
    class FastScanner
    {
        private readonly byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        private readonly Stream stream;

        public FastScanner(Stream stream = null)
        {
            this.stream = stream ?? Console.OpenStandardInput();
        }

        private int ReadByte()
        {
            if (ptr >= len)
            {
                len = stream.Read(buffer, 0, buffer.Length);
                ptr = 0;
                if (len == 0) return -1;
            }
            return buffer[ptr++];
        }

        public int NextInt()
        {
            int c;
            do { c = ReadByte(); } while (c <= 32 && c >= 0);

            int sign = 1;
            if (c == '-')
            {
                sign = -1;
                c = ReadByte();
            }

            int val = 0;
            while (c > 32 && c >= 0)
            {
                val = val * 10 + (c - '0');
                c = ReadByte();
            }
            return val * sign;
        }

        public string NextToken()
        {
            int c;
            do { c = ReadByte(); } while (c <= 32 && c >= 0);

            var sb = new System.Text.StringBuilder();
            while (c > 32 && c >= 0)
            {
                sb.Append((char)c);
                c = ReadByte();
            }
            return sb.ToString();
        }
    }
}
```

**Explanation**

* For each column `j` (0-based) we keep a 64-bit mask – each bit represents a letter that appears in that column among all names.
* While reading the `n` names we set the appropriate bit: `mask[j] |= 1L << (letterIndex)`.
* After all names are processed, the number of different letters in column `j` is the pop-count of `mask[j]`.
  The `PopCount` routine is a classic bit trick (`x & (x-1)`) and is extremely fast.
* The required answer is the product of these counts for all columns, taken modulo `1 000 000 007`.
* The whole algorithm runs in `O(n · m)` time and uses `O(m)` memory, which is optimal for this problem.

The code follows the same input-output format as the original unoptimised version.
